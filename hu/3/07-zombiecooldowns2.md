        contract ZombieFactory is Ownable {

            event NewZombie(uint zombieId, string name, uint dna);

            uint dnaDigits = 16;
            uint dnaModulus = 10 ** dnaDigits;
            uint cooldownTime = 1 days;

            struct Zombie {
              string name;
              uint dna;
              uint32 level;
              uint32 readyTime;
            }

            Zombie[] public zombies;

            mapping (uint => address) public zombieToOwner;
            mapping (address => uint) ownerZombieCount;

            function _createZombie(string memory _name, uint _dna) internal {
                uint id = zombies.push(Zombie(_name, _dna, 1, uint32(now + cooldownTime))) - 1;
                zombieToOwner[id] = msg.sender;
                ownerZombieCount[msg.sender]++;
                emit NewZombie(id, _name, _dna);
            }

            function _generateRandomDna(string memory _str) private view returns (uint) {
                uint rand = uint(keccak256(abi.encodePacked(_str)));
                return rand % dnaModulus;
            }

            function createRandomZombie(string memory _name) public {
                require(ownerZombieCount[msg.sender] == 0);
                uint randDna = _generateRandomDna(_name);
                randDna = randDna - randDna % 100;
                _createZombie(_name, randDna);
            }

        }

Most módosítsuk a `feedAndMultiply`-t, hogy figyelembe vegye a cooldown időzítőnket.

Visszanézve ezt a függvényt, láthatod, hogy `public`-ká tettük az előző leckében. Egy fontos biztonsági gyakorlat, hogy megvizsgálod az összes `public` és `external` függvényedet, és megpróbálod kitalálni, hogyan lehetne visszaélni velük. Emlékezz — hacsak ezek a függvények nem rendelkeznek olyan módosítóval, mint az `onlyOwner`, bármely felhasználó meghívhatja őket, és bármilyen adatot átadhat nekik.

Újra megvizsgálva ezt a konkrét függvényt, a felhasználó közvetlenül meghívhatná, és bármilyen `_targetDna`-t vagy `_species`-t átadhatna. Ez nem tűnik túl játék-szerűnek — azt akarjuk, hogy kövessék a szabályainkat!

Közelebbről megvizsgálva, ez a függvény csak a `feedOnKitty()` által kell meghívva legyen, így a legegyszerűbb módja ezeknek a kihasználásoknak a megelőzésére, hogy `internal`-lá tesszük.

## Tegyük próbára

1. Jelenleg a `feedAndMultiply` egy `public` függvény. Tegyük `internal`-lá, hogy a szerződés biztonságosabb legyen. Nem akarjuk, hogy a felhasználók bármilyen DNS-sel meghívhassák ezt a függvényt.

2. Tegyük úgy, hogy a `feedAndMultiply` figyelembe vegye a `cooldownTime`-ot. Először, miután megtaláltuk a `myZombie`-t, adjunk hozzá egy `require` állítást, amely ellenőrzi az `_isReady()`-t, és átadja neki a `myZombie`-t. Így a felhasználó csak akkor hajthatja végre ezt a függvényt, ha egy zombi cooldown ideje lejárt.

3. A függvény végén hívjuk meg a `_triggerCooldown(myZombie)`-t, hogy a táplálkozás elindítsa a zombi cooldown idejét.
